#program autoexec
#autostart 10
10 REM nextdev - the Next half of the VS Code push-to-hardware loop.
20 REM Pairs with extra/Send-ToNext.ps1 on the PC. Copy this file
30 REM into /nextzxos/ on the card - NOT the root - so every boot
40 REM re-enters the loop: ZX Next Remote soft-resets the machine when
50 REM it exits, which is exactly what closes this cycle.
60 REM
70 REM   no file waiting -> serve the Listener and WAIT for a push
80 REM   file waiting    -> move it aside (so it runs once) and run it
90 REM
100 REM The PC pushes to /dev/incoming.nex, then calls /forceexit: ZX
110 REM Next Remote leaves listen mode, ends its application and resets,
120 REM and the boot lands back here with a file waiting.
130 REM
140 REM ---- which flavour to wait in -----------------------------
150 REM   1 = Http Bridge  (zxnextremote-httpbridge.nex)
160 REM   2 = NextSync     (zxnextremote-n2n.nex)
170 REM Either one works. Both flavours carry the NextSync Listener,
180 REM and it is menu option 2 in both - pick it on the Next, and
190 REM Unite's HTTP bridge drives this machine through it exactly
200 REM as it drives a .sync5 -listen dot. The flavour only decides
210 REM which transport you ALSO get as a Controler when you are not
220 REM pushing builds. If the one you pick is not on the card, the
230 REM other is tried rather than dead-ending here.
240 LET flavour=1
250 REM -----------------------------------------------------------
260 REM Consume a pushed build, if one is waiting.
262 REM
264 REM Nothing below branches on whether a dot command FAILED: .mv
266 REM does not raise a trappable BASIC error when /dev/incoming.nex
268 REM is missing (checked on hardware), so the old "the .mv errors
270 REM when nothing arrived" branch never fired and the Listener at
272 REM line 400 was unreachable on every boot - the machine just
274 REM printed "pushed build did not start." and halted.
276 REM
278 REM So lean on the one thing that IS certain: a .nexload that
280 REM succeeds NEVER COMES BACK. That cuts both ways, and it is why
282 REM the build is retired HERE, before it is loaded, rather than
284 REM after: nothing placed after line 325 can ever run when the
286 REM build actually starts, so a run.nex cleaned up "afterwards"
288 REM is never cleaned up at all - it just re-ran the same game at
290 REM every boot until the card was edited by hand.
292 REM
294 REM Retiring first makes run.nex hold ONLY a freshly pushed build:
296 REM   last boot's build -> /dev/last.nex   (kept, not destroyed:
298 REM                                         .nexload /dev/last.nex
300 REM                                         re-runs it by hand)
302 REM   a new push        -> /dev/run.nex   (and is run below)
304 REM So a push runs exactly once, and the boot after it lands back
306 REM on the Listener ready for the next .ps1 push.
308 REM
309 REM (No colon-tail on the dot lines - the argument stays exactly
310 REM the filename.)
312 ON ERROR GO TO 316
314 .mv /dev/run.nex /dev/last.nex
316 ON ERROR GO TO 320
318 .mv /dev/incoming.nex /dev/run.nex
320 ON ERROR GO TO 400
325 .nexload /dev/run.nex
330 REM Still here, so no build ran: either none was waiting, or one
332 REM was and would not start - and since a dot's failure is not
334 REM reliably trappable we cannot tell the two apart, so the line
336 REM below must read true either way (.nexload prints its own
338 REM reason above when there was a real one).
340 PRINT "nextdev: no build ran - serving the Listener."
400 REM Nothing waiting: hand the machine to the Listener and wait.
410 PRINT "nextdev: waiting for a push..."
420 PRINT "pick 2 NextSync (Listener) on the menu."
430 ON ERROR GO TO 500
440 IF flavour=2 THEN GO TO 470
450 .nexload /dev/zxnextremote-httpbridge.nex
460 GO TO 500
470 .nexload /dev/zxnextremote-n2n.nex
480 GO TO 500
500 REM The preferred flavour did not start - almost always because
510 REM that .nex is not in /dev/. Try the other one before giving
520 REM up, so a card carrying only one flavour still loops.
530 ON ERROR GO TO 600
540 IF flavour=2 THEN GO TO 570
550 .nexload /dev/zxnextremote-n2n.nex
560 GO TO 600
570 .nexload /dev/zxnextremote-httpbridge.nex
580 GO TO 600
600 REM Neither flavour started: say so rather than spinning
610 REM invisibly, and name the two paths that were tried.
620 ON ERROR STOP
630 PRINT "nextdev: no ZX Next Remote found in /dev/."
640 PRINT "Tried zxnextremote-httpbridge.nex"
650 PRINT "  and zxnextremote-n2n.nex"
660 STOP
